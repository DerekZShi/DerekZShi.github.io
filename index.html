        /**
        * @param {string} s
        * @return {boolean}
        */
        var isValid = function(s) {
            let x = 0
            let y = 0
            let z = 0
            let mostRecent = -1
            for (let i=0; i < s.length; i++)
            {
                
                if (s[i] == '{')
                {
                    x++
                    mostRecent = 0
                }
                else if (s[i] == '}')
                {
                    x--
                    if (mostRecent != 0)
                    {
                        return 0
                    }
                }
                else if (s[i] == '(')
                {
                    y++
                    mostRecent = 1
                }
                else if (s[i] == ')')
                {
                    y--
                    if (mostRecent != 1)
                    {
                        return 0
                    }
                }
                else if (s[i] == '[')
                {
                    z++
                    mostRecent = 2
                }
                else if (s[i] == ']')
                {
                    z--
                    if (mostRecent != 2)
                    {
                        console.log('most recent was not [')
                        return 0
                    }
                }
            }
            if (x == 0 && y == 0 && z == 0)
            {
                return 1
            }
            else
            {
                console.log("things do not match")
                return 0
            }
        };
